```kotlin
// engine/model/Budget.kt
enum class BudgetPeriod { DAILY, CYCLE }
data class AppBudget(val id: Long, val packageName: String, val limitBytes: Long, val period: BudgetPeriod, val planId: Long?,
                     val warnAtPercent: Int = 80, val enabled: Boolean = true)
enum class BudgetState { OK, WARNING, EXCEEDED }
data class BudgetStatus(val budget: AppBudget, val usedBytes: Long, val state: BudgetState)

// engine/model/Spike.kt  (fields exactly as contract 11.1)
data class SpikeEvent(val id: Long, val packageName: String, val windowStart: Instant, val detectedAt: Instant, val windowMinutes: Int,
                      val bytes: Long, val baselineBytes: Long, val multiple: Float, val dismissed: Boolean)

// engine/budget/BudgetCalculator.kt
object BudgetCalculator {
    fun state(budget: AppBudget, usedBytes: Long): BudgetState {
        if (budget.limitBytes <= 0L) return BudgetState.OK
        if (usedBytes > budget.limitBytes) return BudgetState.EXCEEDED
        return if (usedBytes * 100 >= budget.limitBytes * budget.warnAtPercent) BudgetState.WARNING else BudgetState.OK
    }

    fun period(budget: AppBudget, now: Instant, zone: ZoneId, planWindow: CycleWindow?): CycleWindow {
        val today = now.atZone(zone).toLocalDate()
        return when (budget.period) {
            BudgetPeriod.DAILY -> CycleWindow(today.atStartOfDay(zone).toInstant(), today.plusDays(1).atStartOfDay(zone).toInstant())
            BudgetPeriod.CYCLE -> planWindow ?: today.withDayOfMonth(1).let {
                CycleWindow(it.atStartOfDay(zone).toInstant(), it.plusMonths(1).atStartOfDay(zone).toInstant())
            }
        }
    }
}

// engine/spike/SpikeDetector.kt
object SpikeDetector {
    private const val FLOOR_BYTES = 1_000_000L
    fun detect(
        recent: List<UsageSlice>, baselinePerWindowBytes: Map<String, Long>, detectedAt: Instant,
        windowMinutes: Int = 120, minBytes: Long = 100_000_000L, minMultiple: Float = 5f,
    ): List<SpikeEvent> = recent.mapNotNull { slice ->
        val pkg = slice.packageName ?: return@mapNotNull null
        val baseline = baselinePerWindowBytes[pkg] ?: return@mapNotNull null
        val multiple = slice.bytes.toFloat() / maxOf(baseline, FLOOR_BYTES)
        if (slice.bytes < minBytes || multiple < minMultiple) null
        else SpikeEvent(0L, pkg, slice.start, detectedAt, windowMinutes, slice.bytes, baseline, multiple, false)
    }
}
```

```kotlin
object HourCurves {
    fun personal(minuteRows: List<MinuteTotal>, now: Instant, zone: ZoneId, base: DoubleArray = ForecastEngine.DEFAULT_HOUR_CURVE): DoubleArray {
        val baseSum = base.sum()
        val normalisedBase = DoubleArray(24) { base[it] / baseSum }
        val today = now.atZone(zone).toLocalDate()
        val from = today.minusDays(7).atStartOfDay(zone).toInstant()
        val to = today.atStartOfDay(zone).toInstant()
        val perHour = DoubleArray(24)
        val days = HashSet<LocalDate>()
        for (row in minuteRows) {
            if (row.network != NetworkKind.MOBILE || row.minuteStart < from || row.minuteStart >= to || row.totalBytes <= 0L) continue
            val local = row.minuteStart.atZone(zone)
            perHour[local.hour] += row.totalBytes.toDouble()
            days += local.toLocalDate()
        }
        val total = perHour.sum()
        if (days.size < 3 || total <= 0.0) return normalisedBase
        val weight = minOf(days.size, 7) / 7.0
        return DoubleArray(24) { h -> weight * perHour[h] / total + (1 - weight) * normalisedBase[h] }
    }
}
```

```kotlin
internal object SchemaFile {
    private const val DIR = "schemas/io.github.khaledbahaaeldin.emberbyte.data.db.EmberbyteDatabase"

    /** Every statement needed to create the database of [version], read from the exported schema (the test working directory is the module directory). */
    fun statements(version: Int): List<String> {
        val db = JSONObject(File("$DIR/$version.json").readText()).getJSONObject("database")
        val out = mutableListOf<String>()
        val entities = db.getJSONArray("entities")
        for (i in 0 until entities.length()) {
            val entity = entities.getJSONObject(i)
            val table = entity.getString("tableName")
            out += entity.getString("createSql").replace("\${TABLE_NAME}", table)
            entity.optJSONArray("indices")?.let { indices ->
                for (j in 0 until indices.length()) out += indices.getJSONObject(j).getString("createSql").replace("\${TABLE_NAME}", table)
            }
        }
        db.optJSONArray("setupQueries")?.let { setup -> for (i in 0 until setup.length()) out += setup.getString(i) }
        return out
    }
}
```

```kotlin
@OptIn(ExperimentalCoroutinesApi::class)
class RoomBudgetRepository(
    private val budgets: BudgetStore, private val plans: PlanStore, private val usage: UsageStore,
    private val clock: Clock, private val ticks: Flow<Instant> = minuteTicks(clock),
) : BudgetRepository {
    private data class Period(val window: CycleWindow, val subscriptionId: Int?)

    private fun periodFor(budget: AppBudget, planList: List<Plan>, now: Instant): Period {
        val plan = if (budget.period == BudgetPeriod.CYCLE) planList.firstOrNull { it.id == budget.planId } else null
        val planWindow = plan?.let { CycleCalculator.windowAt(it.cycle, now) }
        return Period(BudgetCalculator.period(budget, now, clock.zone, planWindow), plan?.subscriptionId)
    }

    private fun uidsOf(packageName: String, meta: List<AppMeta>): Set<Int> {
        val named = meta.filter { it.packageName == packageName }.map { it.uid }.toSet()
        return if (named.isNotEmpty() || !packageName.startsWith("uid:")) named else setOfNotNull(packageName.removePrefix("uid:").toIntOrNull())
    }

    private fun snapshot(now: Instant): Flow<List<BudgetPeriodStatus>> =
        combine(budgets.observeBudgets(), plans.observePlans(), usage.observeAppMeta()) { b, p, m -> Triple(b, p, m) }
            .flatMapLatest { (list, planList, meta) ->
                if (list.isEmpty()) flowOf(emptyList())
                else {
                    val periods = list.map { it to periodFor(it, planList, now) }
                    val from = periods.minOf { it.second.window.start }.minusSeconds(SeriesBuilder.WINDOW_SECONDS)
                    usage.observeHourlyRows(from, OPEN_END).map { rows ->
                        periods.map { (budget, period) ->
                            val uids = uidsOf(budget.packageName, meta)
                            val used = rows.filter {
                                it.uid in uids && it.network == NetworkKind.MOBILE &&
                                    it.hourStart >= period.window.start && it.hourStart < period.window.end &&
                                    (period.subscriptionId == null || it.subscriptionId == period.subscriptionId)
                            }.sumOf { it.totalBytes }
                            BudgetPeriodStatus(
                                BudgetStatus(budget, used, BudgetCalculator.state(budget, used)), period.window.start,
                                meta.firstOrNull { it.packageName == budget.packageName }?.label ?: budget.packageName,
                            )
                        }
                    }
                }
            }

    override fun observeStatuses(): Flow<List<BudgetStatus>> = ticks.flatMapLatest { snapshot(it) }.map { list -> list.map { it.status } }
    suspend fun currentPeriodStatuses(): List<BudgetPeriodStatus> = snapshot(clock.instant()).first()
    // upsert / delete: validation as in contract 11.2, the same try/catch + CancellationException pattern as RoomPlanRepository.guarded
}
```

```kotlin
// BudgetEvaluator.kt
class BudgetEvaluator(private val statuses: suspend () -> List<BudgetPeriodStatus>, private val store: BudgetStore) {
    private val mutex = Mutex()
    suspend fun evaluate(): List<BudgetNotice> = mutex.withLock {
        val notices = ArrayList<BudgetNotice>()
        for (entry in statuses()) {
            val budget = entry.status.budget
            if (!budget.enabled) continue
            val last = store.lastState(budget.id, entry.periodStart) ?: BudgetState.OK
            val state = entry.status.state
            if (state.ordinal > last.ordinal) notices += BudgetNotice(budget, entry.label, entry.status.usedBytes, state)
            if (state != last) store.setState(budget.id, entry.periodStart, state)
        }
        notices
    }
}
```

```kotlin
// SpikeEvaluator.kt
class SpikeEvaluator(private val usage: UsageStore, private val store: SpikeStore, private val clock: Clock) {
    private val mutex = Mutex()
    private val window = Duration.ofSeconds(SeriesBuilder.WINDOW_SECONDS)

    suspend fun evaluate(): List<SpikeEvent> = mutex.withLock {
        val now = clock.instant()
        val current = SeriesBuilder.windowStart(now)
        val windows = listOf(current, current.minus(window), current.minus(window.multipliedBy(2)))
        val historyStart = windows.last().minus(Duration.ofDays(HISTORY_DAYS)).minus(window)
        val allRows = usage.hourlyRows(historyStart, current.plus(window))
        val packageByUid = usage.appMeta().associate { it.uid to it.packageName }
        val firstSeen = HashMap<String, Instant>()
        val mobileBytes = HashMap<Pair<String, Instant>, Long>()
        for (row in allRows) {
            val pkg = packageByUid[row.uid] ?: continue
            firstSeen.merge(pkg, row.hourStart) { a, b -> minOf(a, b) }
            if (row.network == NetworkKind.MOBILE) mobileBytes.merge(pkg to row.hourStart, row.totalBytes) { a, b -> a + b }
        }
        val stored = ArrayList<SpikeEvent>()
        for (start in windows) {
            val candidates = mobileBytes.filterKeys { it.second == start }.map { (key, bytes) -> key.first to bytes }
            val eligible = candidates.filter { (pkg, _) -> firstSeen.getValue(pkg) <= start.minus(Duration.ofDays(MIN_HISTORY_DAYS)) }
            val slices = eligible.map { (pkg, bytes) -> UsageSlice(start, start.plus(window), pkg, bytes) }
            val baselines = eligible.associate { (pkg, _) -> pkg to median(pkg, start, mobileBytes) }
            for (event in SpikeDetector.detect(slices, baselines, now)) {
                if (store.insertIfAbsent(event)) stored += event
            }
        }
        stored
    }

    /** Median of the app's mobile bytes in the 168 windows before [start] (windows without a row count as 0). */
    private fun median(pkg: String, start: Instant, bytes: Map<Pair<String, Instant>, Long>): Long {
        val values = LongArray(HISTORY_DAYS.toInt() * 12) { i -> bytes[pkg to start.minus(window.multipliedBy(i + 1L))] ?: 0L }
        values.sort()
        return (values[values.size / 2 - 1] + values[values.size / 2]) / 2
    }

    /** Applies the notification rate limit (one per app per hour) and remembers the notified apps. Contract 11.2. */
    suspend fun selectForNotification(events: List<SpikeEvent>): List<SpikeEvent> = mutex.withLock {
        val now = clock.instant()
        val chosen = ArrayList<SpikeEvent>()
        val handled = HashSet<String>()
        for (event in events) {
            if (!handled.add(event.packageName)) continue
            val last = store.lastNoticeAt(event.packageName)
            if (last != null && Duration.between(last, now) < Duration.ofHours(1)) continue
            store.markNotified(event.packageName, now)
            chosen += event
        }
        chosen
    }

    private companion object { const val HISTORY_DAYS = 14L; const val MIN_HISTORY_DAYS = 3L }
}
```

```kotlin
object BackupCodec {
    const val MAGIC = "EMBBKP01"
    private const val SALT = 16; private const val NONCE = 12; private const val HEADER = 8 + SALT + 12 + NONCE      // 48
    data class Params(val memoryKiB: Int = 65_536, val iterations: Int = 3, val parallelism: Int = 1)
    const val MAX_MEMORY_KIB = 262_144; const val MAX_ITERATIONS = 10; const val MAX_PARALLELISM = 4

    fun seal(plain: ByteArray, passphrase: CharArray, params: Params = Params(), random: SecureRandom = SecureRandom()): ByteArray {
        val salt = ByteArray(SALT).also(random::nextBytes)
        val nonce = ByteArray(NONCE).also(random::nextBytes)
        val header = ByteBuffer.allocate(HEADER).put(MAGIC.toByteArray(Charsets.US_ASCII)).put(salt)
            .putInt(params.memoryKiB).putInt(params.iterations).putInt(params.parallelism).put(nonce).array()
        val key = deriveKey(passphrase, salt, params)
        try {
            val cipher = Cipher.getInstance("AES/GCM/NoPadding")
            cipher.init(Cipher.ENCRYPT_MODE, SecretKeySpec(key, "AES"), GCMParameterSpec(128, nonce))
            cipher.updateAAD(header)
            return header + cipher.doFinal(plain)
        } finally { key.fill(0) }
    }

    fun open(sealed: ByteArray, passphrase: CharArray): ByteArray {
        if (sealed.size < HEADER + 16) throw BackupFormatException("file", "This is not an Emberbyte backup")
        if (String(sealed, 0, 8, Charsets.US_ASCII) != MAGIC) throw BackupFormatException("file", "This is not an Emberbyte backup")
        val buffer = ByteBuffer.wrap(sealed, 8, HEADER - 8)
        val salt = ByteArray(SALT).also(buffer::get)
        val params = Params(buffer.int, buffer.int, buffer.int)
        if (params.memoryKiB !in 8..MAX_MEMORY_KIB || params.iterations !in 1..MAX_ITERATIONS || params.parallelism !in 1..MAX_PARALLELISM) {
            throw BackupFormatException("file", "The backup uses settings this version does not accept")
        }
        val nonce = ByteArray(NONCE).also(buffer::get)
        val key = deriveKey(passphrase, salt, params)
        try {
            val cipher = Cipher.getInstance("AES/GCM/NoPadding")
            cipher.init(Cipher.DECRYPT_MODE, SecretKeySpec(key, "AES"), GCMParameterSpec(128, nonce))
            cipher.updateAAD(sealed, 0, HEADER)
            return cipher.doFinal(sealed, HEADER, sealed.size - HEADER)
        } catch (_: AEADBadTagException) {
            throw BackupFormatException("passphrase", "Wrong passphrase or damaged file")
        } catch (_: GeneralSecurityException) {
            throw BackupFormatException("passphrase", "Wrong passphrase or damaged file")
        } finally { key.fill(0) }
    }

    private fun deriveKey(passphrase: CharArray, salt: ByteArray, params: Params): ByteArray {
        val generator = Argon2BytesGenerator()
        generator.init(
            Argon2Parameters.Builder(Argon2Parameters.ARGON2_id).withVersion(Argon2Parameters.ARGON2_VERSION_13)
                .withSalt(salt).withMemoryAsKB(params.memoryKiB).withIterations(params.iterations).withParallelism(params.parallelism).build(),
        )
        return ByteArray(32).also { generator.generateBytes(passphrase, it) }
    }
}
```

```kotlin
internal fun NavController.openDeepLink(intent: Intent, allowed: () -> Boolean = { true }): Boolean {
    val uri = intent.data ?: return false
    if (!allowed()) return false
    val target = DeepLinkTarget.parse(uri) ?: return false
    val current = currentBackStackEntry
    return try {
        when (target) {
            is DeepLinkTarget.Tab -> navigate(target.destination.route) {          // the same options as the navigation bar
                popUpTo(Destination.Home.route) { saveState = true }; launchSingleTop = true; restoreState = true
            }
            DeepLinkTarget.Settings -> if (current?.destination?.route != Routes.SETTINGS) navigate(uri)
            is DeepLinkTarget.AppDetail ->
                if (!(current?.destination?.route == Routes.APP_DETAIL && current.arguments?.getString("packageName") == target.packageName)) navigate(uri)
            is DeepLinkTarget.PlanEditor ->
                if (!(current?.destination?.route == Routes.PLAN_EDITOR && current.arguments?.getLong("planId") == target.planId)) navigate(uri)
        }
        true
    } catch (_: IllegalArgumentException) { false }
}
```
